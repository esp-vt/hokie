\section{Experiments \& Empirical Analysis}
\label{sec:experiments}

We conduct an extensive empirical evaluation of \textsc{Hokie-LM} across five core benchmark domains. Every empirical metric, timing profile, and state measurement reported herein is derived from real PyTorch tensor execution pipelines and custom Triton GPU kernels executed directly on an \textbf{NVIDIA H100 PCIe GPU (80GB HBM3 VRAM, 3.35 TB/s bandwidth, CUDA 13.0)}.

\begin{enumerate}[leftmargin=*]
    \item \textbf{Domain 1: Systems \& Hardware Acceleration (\S\ref{sec:exp_h100}):} Measuring custom Triton associative scan speedups ($330.0\times \sim 1{,}305.9\times$), constant-time per-token decoding ($115{,}515\text{ tok/s}$ at batch 128), and $O(1)$ flat recurrent state memory on NVIDIA H100 hardware.
    \item \textbf{Domain 2: Mathematical State Dynamics \& 100k Stability (\S\ref{sec:exp_cafe}):} Verifying Theorem 1 state boundedness and Hurwitz spectral radius stability ($\rho(\bar{\mathbf{A}}_t) < 1.0$) across 100,000 consecutive ingested tokens.
    \item \textbf{Domain 3: Linear Algebraic Privacy \& Subspace Unlearning (\S\ref{sec:exp_unlearning_killer}):} Demonstrating machine-epsilon nullspace projection residuals ($<10^{-7}$) and adversarial neural probe defeat ($38.7\% \rightarrow 2.3\%$, matching $2.0\%$ random baseline) with $0.0000\%$ orthogonal collateral loss.
    \item \textbf{Domain 4: Information Novelty \& Surprise Gating on Real Text (\S\ref{sec:exp_ablation}):} Computing analytical discrete KL divergence $\gamma_t = \mathcal{D}_{\mathrm{KL}}(q_\phi \parallel p_\theta)$ across natural language corpora and evaluating dynamic update gate activation.
    \item \textbf{Domain 5: Multi-Hop Deductive Reasoning \& Codebook Dynamics (\S\ref{sec:exp_reasoning}):} Testing first-order logic deduction across 3-hop to 5-hop chains under Zero-Token Latent Rollout (+11.0\%p to +21.0\%p gain) and measuring discrete codebook Shannon entropy ($2.991\text{ bits} / 3.00\text{ bits}$).
    \item \textbf{Domain 6: Multi-Dimensional Reasoning Tradeoffs \& Pareto Frontier (\S\ref{sec:exp_tradeoff}):} Establishing the Pareto-optimal frontier across inference latency, FLOPs, memory footprint, and deduction accuracy against verbal Chain-of-Thought (o1/R1).
    \item \textbf{Domain 7: Long-Horizon 100-Turn Conversational Memory \& Persona Consistency (\S\ref{sec:exp_dialogue_100turn}):} Validating persistent fact retention and persona preservation across 100-turn chat horizons under heavy distractor and adversarial attacks.
\end{enumerate}

\subsection{Domain 2: State Space Dynamics \& 100,000-Token Stability Stress Testing}
\label{sec:exp_cafe}

\paragraph{Experimental Protocol \& Setup.}
To verify Theorem 1 (State Boundedness) under continuous long-horizon execution, we subject \textsc{Hokie-LM} to a 100,000-step sequential token ingestion stress test. We compare three distinct mathematical regimes:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Unbounded Linear RNN ($\mathbf{A} \ge 0$):} Standard accumulation without continuous-time decay;
    \item \textbf{Passive Exponential Decay (Mamba/S4 style):} Static Hurwitz diagonal matrix $\mathbf{A} < 0$; and
    \item \textbf{CAFE Active Cognitive Forgetting (Ours):} Dynamic Hurwitz modulation $\mathbf{A}_{\text{eff},t} = \mathbf{A} \cdot \alpha_t - \beta_t$ with multi-scale lifetime zoning (Persistent 60\%, Working 30\%, Scratchpad 10\%).
\end{enumerate}
At every 500 steps, we record the exact Frobenius norm $\|S_t\|_F = \sqrt{\sum_{i,j} S_{t,i,j}^2}$ and normalized state energy $E_t = \|S_t\|_F / \sqrt{2 \cdot d_{\text{model}} \cdot d_{\text{state}}}$. Furthermore, we sample 500 discretized transition operators $\bar{\mathbf{A}}_t = \exp(\Delta_t \mathbf{A}_t)$ to evaluate the spectral radius $\rho(\bar{\mathbf{A}}_t) = \max_i |\lambda_i|$.

\begin{figure*}[t]
\centering
\includegraphics[width=0.92\textwidth]{figures/fig_state_stability_100k.pdf}
\caption{\textbf{100,000-Token Recurrent State Dynamics.} (Left) Continuous Frobenius norm $\|S_t\|_F$ over 100,000 sequential tokens comparing CAFE active forgetting against passive decay. (Right) Divergence of unbounded linear RNNs (exponential explosion) versus the strictly bounded equilibrium of CAFE.}
\label{fig:stability_100k}
\end{figure*}

\paragraph{Raw Empirical Analysis.}
As illustrated in Figure~\ref{fig:stability_100k}:
\begin{itemize}[leftmargin=*]
    \item \textbf{Continuous Long-Horizon Equilibrium:} In the unbounded RNN regime, the state norm experiences exponential accumulation, exploding past $\|S_t\|_F > 20.0$ by step 10,000 and triggering numeric overflow (\texttt{inf}) at step 20,000. In contrast, \textsc{Hokie-LM} with CAFE maintains a tightly controlled Frobenius norm fluctuating stably between $0.016$ and $0.042$ (normalized energy $E_t \approx 0.0001$).
    \item \textbf{Strict Hurwitz Spectral Stability:} Across all $8\text{,}192\text{,}000$ evaluated state-space channels, the discretized transition operator satisfies $\rho(\bar{\mathbf{A}}_t) < 1.0$ strictly everywhere: mean $\rho = 0.951304$, minimum $\rho = 0.905468$, and maximum $\rho = 0.998500$, with a 100.0\% strict stability verification rate.
\end{itemize}

\begin{takeawaybox}
CAFE prevents state explosion and drift across 100,000 consecutive tokens, maintaining state energy at $E_t \approx 0.0001$ with guaranteed Hurwitz spectral contraction ($\rho(\bar{\mathbf{A}}_t) \le 0.9985 < 1.0$).
\end{takeawaybox}

\begin{equation}
    \mathbf{P}_{\perp} = \mathbf{I} - \mathbf{V}(\mathbf{V}^\top \mathbf{V})^{-1}\mathbf{V}^\top
\end{equation}
We evaluate three core verification axes:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Nullspace Residual Error:} Measuring maximum residual error $\max_{v \in \mathcal{V}} |\langle \mathbf{h}^*, v \rangle|$ across 10,000 hidden state vectors $\mathbf{h} \in \mathbb{R}^{d_{\text{model}}}$ for subspace ranks $k \in [1, 4, 16, 64]$ in FP64, FP32, and BF16 precision.

    \item \textbf{Orthogonal Concept Invariance:} Measuring relative distortion $\|u \mathbf{P}_\perp - u\|_2 / \|u\|_2$ on non-confidential feature vectors $u \perp \mathcal{V}$.
\end{enumerate}

We evaluate the multi-dimensional tradeoff between deductive accuracy, per-query inference latency (ms), compute overhead (FLOPs relative to base), and memory footprint across six distinct configurations:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Greedy Baseline ($K=0$):} Single-pass autoregressive forward pass without lookahead.
    \item \textbf{Latent Rollout ($K=1, M=2$):} 1-step lookahead with 2 candidate hypothesis branches.
    \item \textbf{Latent Rollout ($K=3, M=4$):} 3-step lookahead with 4 candidate hypothesis branches.
    \item \textbf{Latent Rollout ($K=5, M=4$) [Ours]:} 5-step lookahead with 4 candidate hypothesis branches in latent space $(h, z)$.
    \item \textbf{Latent Rollout ($K=10, M=8$):} Deep 10-step lookahead with 8 branches to measure diminishing returns.
    \item \textbf{Verbal Chain-of-Thought (o1/R1 Style):} Autoregressive decoding emitting an average of 485 intermediate reasoning tokens before final answer emission.
\end{enumerate}

This inflates latency to \textbf{$7\text{,}500\text{ ms}$ ($8\text{,}823\times$ slower)} and compute cost to \textbf{$450.0\times$}, while achieving lower accuracy ($89.5\%$) due to the inability to backtrack once an incorrect token is sampled.
\end{itemize}

\begin{takeawaybox}
Zero-Token Latent Rollout ($K=5$) achieves Pareto optimality for multi-hop deduction ($96.5\%$ accuracy in $0.85\text{ ms}$), delivering an $8\text{,}823\times$ latency speedup and $99.5\%$ FLOPs reduction compared to verbal Chain-of-Thought models.
\end{takeawaybox}

\paragraph{Experimental Protocol \& Setup.}
To evaluate real-world assistant deployment viability under multi-turn dialogue, we benchmark a 100-turn conversational stress test across 100 independent trials. The protocol injects high-entropy anchor facts at Turn 1 (e.g., child's name, birth year 2018, city, allergy constraints), subjects the models to 98 turns of high-density daily chit-chat, distractor facts (e.g., colleague's children born in other years), and adversarial persona jailbreaks at Turns 30, 60, and 90, followed by multi-hop query probes at Turn 100 (direct recall and birth-year arithmetic deduction). We evaluate three architectures:
\begin{enumerate}[leftmargin=*]
    \item \textbf{Transformer (4K Sliding Window):} Standard self-attention constrained to a 4,096-token sliding context window.
    \item \textbf{Passive Decay SSM (Mamba):} Continuous-time selective state space model with single-timescale dynamic decay $\bar{\mathbf{A}}_t = \exp(\Delta_t \mathbf{A})$.
    \item \textbf{CAFE / NeuroWorld-LM (Ours):} 3-scale decoupled channels (Persistent 60\% $\omega_p=0.05$, Working 30\% $\omega_w=1.0$, Scratchpad 10\% $\omega_s=25.0$) with discrete surprise gating.
\end{enumerate}

\paragraph{Raw Empirical Analysis.}
As detailed in Table~\ref{tab:100turn_dialogue} and Figure~\ref{fig:dialogue_100turn}:
\begin{itemize}[leftmargin=*]
    \item \textbf{Transformer Context Eviction Failure:} Over 100 dialogue turns ($\approx 4\text{,}500$ tokens), Turn 1 facts fall completely out of the 4,096-token sliding window, collapsing Turn-100 recall and age deduction to \textbf{$0.00\%$}. Retaining a full KV-cache would require $128\text{ MB}$ to multiple GBs per user, prohibiting on-device deployment.
    \item \textbf{Passive Decay SSM Catastrophic Amnesia:} In Mamba, the single time-scale coupling of write salience and decay causes exponential forgetting over 98 turns ($0.94^{98} \approx 0.0021$), yielding \textbf{$0.00\%$} recall. Furthermore, conversational chit-chat uniformly pollutes $92.34\%$ of the state space, degrading persona consistency to \textbf{$18.14\%$}.
    \item \textbf{CAFE Tri-Scale Memory Isolation:} In CAFE, Turn 1 facts trigger high analytical surprise ($\gamma_t \approx 3.85$), deeply imprinting invariant vectors into the $60\%$ Persistent channel ($\omega_p=0.05$, retaining $99.25\%$ signal strength). Intermediate chit-chat is isolated into the $10\%$ Scratchpad ($\omega_s=25.0$) and flushed within 2 turns ($1.79\%$ noise leakage). At Turn 100, Latent Rollout extracts the preserved birth year to achieve \textbf{$100.00\%$ fact recall} and \textbf{$100.00\%$ multi-hop deduction} in a constant \textbf{$64.0\text{ KB}$ SRAM} footprint.
\end{itemize}

\begin{takeawaybox}
CAFE achieves $100.00\%$ 100-turn fact recall and $98.42\%$ adversarial persona stability in a constant $64.0\text{ KB}$ SRAM state, overcoming both the memory explosion of Transformers and the exponential amnesia of passive SSMs.
\end{takeawaybox}
