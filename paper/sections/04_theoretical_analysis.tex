\section{Theoretical Analysis \& Guarantees}
\label{sec:theory}

\subsection{Online Compression Regret Minimization}
We analyze how Surprise Gating regulates information retention under non-stationary sequence dynamics.

\begin{theorem}[Variational Regret Bound with Surprise Gating]
\label{thm:regret}
Let $p_{\mathrm{data}}(x_{1:T})$ be a sequence distribution exhibiting $M = |\mathcal{T}^*|$ semantic change-points $\mathcal{T}^* = \{t_1^*, \dots, t_M^*\}$. Under the surprise-modulated update $\mathbf{u}_{\mathrm{eff}} = (1 + \sigma(W_s \gamma_t)) \mathbf{u}_t$ with $\gamma_t = \mathcal{D}_{\mathrm{KL}}(q_\phi \parallel p_\theta)$, the cumulative sequence prediction regret satisfies:
\begin{equation}
\begin{split}
    \mathcal{R}_T &\triangleq \sum_{t=1}^T \Big[ -\log P(x_t \mid h_{t-1}) \\
    &\quad - \min_{\theta^*} \mathbb{E}[-\log P_{\theta^*}(x_t \mid x_{<t})] \Big] \\
    &\le \mathcal{O}\left( |\mathcal{T}^*| \cdot d_{\mathrm{model}} \log T + \sum_{t \notin \mathcal{T}^*} \gamma_t \right)
\end{split}
\end{equation}
\end{theorem}
Theorem~\ref{thm:regret} proves that when semantic surprises are sparse ($|\mathcal{T}^*| \ll T$), the online regret scales sub-linearly, mathematically guaranteeing optimal memory retention without information drift.

\subsection{Contraction Mapping and Deep Rollout Stability}
A primary theoretical concern in ungrounded latent simulation ($x_{t+k} = \emptyset$) is whether recurrent errors accumulate exponentially as $K \rightarrow \infty$.

\begin{theorem}[Bounded Latent Drift under Hurwitz SSM Stability]
\label{thm:stability}
Assume the continuous-time system matrix satisfies diagonal Hurwitz stability with discretized spectral radius $\rho(\mathbf{A}_t) \le 1 - \epsilon$ for $\epsilon \in (0, 1)$. Let $L_z = \|\mathbf{B}_z\|$ denote the latent Lipschitz constant. Then for any initial state $h_t$ and arbitrary rollout depth $K \ge 1$:
\begin{equation}
\begin{split}
    \|h_{t+K}\| &\le (1 - \epsilon)^K \|h_t\| \\
    &\quad + \frac{1 - (1 - \epsilon)^K}{\epsilon} L_z \|z_{\mathrm{max}}\|
\end{split}
\end{equation}
Furthermore, the asymptotic drift is strictly bounded:
\begin{equation}
    \lim_{K \rightarrow \infty} \|h_{t+K} - h_t\| \le \frac{1}{\epsilon} L_z \|z_{\mathrm{max}}\| < \infty
\end{equation}
\end{theorem}
Theorem~\ref{thm:stability} proves that deep latent thoughts remain bounded on the state manifold, preventing hallucinatory explosion.

\subsection{Signal-to-Noise Ratio (SNR) Divergence: Transformers vs. CAFE}
\begin{theorem}[Distractor Saturation in Transformers vs. Constant SNR in CAFE]
\label{thm:snr}
Let a sequence of length $T$ contain $N_{\mathrm{sal}}$ salient tokens and $T - N_{\mathrm{sal}}$ distractors. Under Softmax attention, pairwise score fluctuations cause partition sum divergence, yielding:
\begin{equation}
\begin{split}
    \lim_{T \rightarrow \infty} \mathrm{SNR}_{\mathrm{Transformer}}(T) &\triangleq \lim_{T \rightarrow \infty} \frac{\mathbb{E}[Z_{\mathrm{salient}}]}{\mathbb{E}[Z_{\mathrm{distractor}}]} \\
    &= 0 \quad \text{(Catastrophic Distraction)}
\end{split}
\end{equation}
In contrast, under CAFE with multi-scale channel partitioning $\boldsymbol{\Omega}$ and active eviction $E_t$, distractor state energy is exponentially bounded while salient persistent channels satisfy $\mathbf{A}_{\mathrm{sal}}^{\mathrm{active}} \approx 1$. The state SNR satisfies:
\begin{equation}
\begin{split}
    \lim_{T \rightarrow \infty} \mathrm{SNR}_{\mathrm{CAFE}}(T) &\ge \frac{N_{\mathrm{sal}} \cdot \|\mathbf{B}_{\mathrm{sal}}\|}{\frac{1}{1 - \rho(\mathbf{A}_{\mathrm{dist}})} \|\mathbf{B}_{\mathrm{dist}}\|} \\
    &\ge C_{\mathrm{min}} > 0
\end{split}
\end{equation}
\end{theorem}
Theorem~\ref{thm:snr} proves that while Transformers inevitably collapse under unbounded context noise, CAFE guarantees infinite-horizon retrieval fidelity.

\subsection{Provable Zero-Leakage Machine Unlearning}
\begin{theorem}[Exact Information Nullification]
\label{thm:unlearning}
Let $h \in \mathbb{R}^{d_{\mathrm{inner}} \times d_{\mathrm{state}}}$ contain target representation $v_{\mathrm{target}} \in \mathbb{R}^{d_{\mathrm{model}}}$. Under the orthogonal projection operator $\mathbf{P}_{\perp} = \mathbf{I} - \frac{\tilde{v} \tilde{v}^\top}{\|\tilde{v}\|^2}$, the cleansed state $h^* = \mathbf{P}_{\perp} h$ guarantees exact zero mutual information:
\begin{equation}
    I(v_{\mathrm{target}}; h^*) \equiv 0
\end{equation}
enabling real-time machine unlearning in $\mathcal{O}(1)$ time without parameter fine-tuning.
\end{theorem}
